# ─────────────────────────── ML-ядро (жёстко зафиксировано) ───────────────────────────
# transformers запинен в requirements.txt на 4.44.2 (см. также setup.sh: XTTS вынесен

# транзитивные пакеты, версии которых обязаны быть совместимы с transformers==4.44.2.

torch>=2.6.0
# Pillow: 11.0.0 уязвим — ядро требует >=11.3.0 (requirements.txt).
Pillow>=11.3.0
